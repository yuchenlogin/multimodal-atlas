\documentclass[11pt,a4paper]{ctexart}

% ---------- 页面几何 ----------
\usepackage{geometry}
\geometry{

  top=2.8cm, bottom=2.8cm,

}

% ---------- 数学 ----------
\usepackage{amsmath}
\usepackage{amssymb}

% ---------- 图形 / 表格 ----------
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\usepackage{caption}
\captionsetup{
  font={small},
  labelfont={bf},
  labelsep=quad,
  skip=6pt
}
% L 列: 等宽 raggedright, 实际宽度由 md2tex.py 按列数生成 p{...}

\renewcommand{\arraystretch}{1.25}

% ---------- 颜色 ----------
\usepackage{xcolor}
\definecolor{accent}{HTML}{1F4E79}   % 深蓝: 章节标题 / 链接
\definecolor{accent2}{HTML}{2E75B6}  % 中蓝: 子标题
\definecolor{softgray}{HTML}{595959} % 页眉 / caption 灰
\definecolor{quotebg}{HTML}{F4F6F8}  % 摘要 / 引文底色
\definecolor{codebg}{HTML}{F6F6F2}   % 代码底色

% ---------- 引文 / 摘要框 (framed 比 tcolorbox 轻量稳定) ----------
\usepackage{framed}
\colorlet{shadecolor}{quotebg}
% 摘要样式
\renewenvironment{abstract}{%
  \begin{snugshade}\small
  \noindent\textbf{\textcolor{accent}{摘要}}\quad\ignorespaces
}{\end{snugshade}\medskip}

% ---------- 章节标题样式 ----------
\usepackage{titlesec}
\titleformat{\section}
  {\Large\bfseries\color{accent}}
  {\thesection}{0.8em}{}
  [{\color{accent!45}\titlerule[0.6pt]}]
\titleformat{\subsection}
  {\large\bfseries\color{accent2}}
  {\thesubsection}{0.8em}{}
\titleformat{\subsubsection}
  {\normalsize\bfseries\color{softgray}}
  {\thesubsubsection}{0.8em}{}
\titlespacing*{\section}{0pt}{1.6em plus .3em}{0.9em}
\titlespacing*{\subsection}{0pt}{1.2em plus .2em}{0.6em}
\titlespacing*{\subsubsection}{0pt}{1.0em plus .2em}{0.4em}

% ---------- 页眉页脚 ----------
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\color{softgray}\nouppercase{\leftmark}}
\fancyhead[R]{\small\color{softgray} 多模态大模型系列报告}
\fancyfoot[C]{\small\color{softgray}\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\headrule}{\color{accent!40}\hrule width\headwidth height 0.4pt}
% section 标记只显示标题文字
\renewcommand{\sectionmark}[1]{\markboth{\thesection\ #1}{}}

% ---------- 列表 ----------
\usepackage{enumitem}
\setlist{
  itemsep=2pt, parsep=2pt, topsep=6pt,

}

\setlist[enumerate,1]{label=\textcolor{accent2}{\arabic*.}}

% ---------- 超链接 ----------
\usepackage[hidelinks]{hyperref}
\hypersetup{

  linkcolor=accent,
  urlcolor=accent2,
  citecolor=accent2,
  pdfencoding=auto,
  bookmarksnumbered=true
}

\urlstyle{same}

% ---------- 杂项排版 ----------
